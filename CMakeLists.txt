cmake_minimum_required(VERSION 3.20)
project(MaterialLab VERSION 2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MATERIAL_LAB_GUI "Build desktop application" ON)
option(MATERIAL_LAB_SANITIZERS "Enable address/undefined sanitizers" OFF)
add_library(material_physics
    src/physics/Material.cpp src/physics/PhysicsWorld.cpp
    src/physics/Collision.cpp src/physics/Deformation.cpp
    src/physics/Fracture.cpp src/physics/Explosion.cpp src/physics/Thermal.cpp
    src/editor/Editor.cpp src/app/SimulationState.cpp src/app/SceneIO.cpp src/app/Map.cpp src/app/Timeline.cpp)
find_package(Threads REQUIRED)
target_link_libraries(material_physics PUBLIC Threads::Threads)
target_include_directories(material_physics PUBLIC src)
target_compile_options(material_physics PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Wpedantic>)
if(MATERIAL_LAB_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(material_physics PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(material_physics PUBLIC -fsanitize=address,undefined)
endif()
if(MATERIAL_LAB_GUI)
    find_package(raylib 5.5 QUIET)
    if(NOT raylib_FOUND)
        set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
        if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/raylib/CMakeLists.txt")
            add_subdirectory(third_party/raylib EXCLUDE_FROM_ALL)
        else()
            include(FetchContent)
            FetchContent_Declare(raylib GIT_REPOSITORY https://github.com/raysan5/raylib.git GIT_TAG 5.5 GIT_SHALLOW TRUE)
            FetchContent_MakeAvailable(raylib)
        endif()
    endif()
    add_executable(material_lab src/main.cpp src/app/Application.cpp src/rendering/Renderer.cpp src/ui/UI.cpp)
    target_link_libraries(material_lab PRIVATE material_physics raylib)
    add_custom_command(TARGET material_lab POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_directory
        "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:material_lab>/assets")
    target_compile_options(material_lab PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Wpedantic>)
endif()
enable_testing()
add_executable(physics_tests tests/PhysicsTests.cpp)
target_link_libraries(physics_tests PRIVATE material_physics)
add_test(NAME physics_validation COMMAND physics_tests)
